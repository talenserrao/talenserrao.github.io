GUI Command and Control (Client + Controller)

I wanted a simple way to send system commands from a GUI and get the output back in real time. The result is a small Python command-and-control demo built on sockets with a tkinter controller. It is intentionally minimal: one GUI, a client that listens for commands, and a few safe system info queries.

This post breaks down the flow, the code, and how it works.

How It Works

The project has two parts:

  • client.py: connects to the server, waits for a command string, runs it, and returns output.
  • controller.py: provides a GUI to select commands, send them to one or more clients, and display responses.

Commands are simple string identifiers like get_users() or get_mem_info(). The client maps those to local commands and sends the output back over the socket.

Client: Command Receiver

The client opens a socket connection and waits for a command. When it receives a known command, it runs a local command with subprocess and sends the output back to the server.

import socket
import subprocess

def client():
    """
    it creates a ocket client that can receive commands from a server and send back system information
    
    """
    try:
        host = 'localhost'
        port = 9999

        sock = socket.socket()
        sock.connect((host, port))

        while True:
            # Receive data from the client
            received = sock.recv(1024).decode()
            if not received:
                break
            print("The data has been received: " + received)

            if received == "get_users()":
                # Run the `cat /etc/passwd` command and send the output to the client
                output = subprocess.check_output(["cat", "/etc/passwd"]).decode().strip()
                sock.send(output.encode())

            elif received == 'get_id()':
                # Run the `id` command and send the output to the client
                output = subprocess.check_output(["id"]).decode().strip()
                sock.send(output.encode())

            elif received == "get_device_info()":
                # Show the device information in /proc/devices and send the output to the client
                output = subprocess.check_output(["cat", "/proc/devices"]).decode().strip()
                sock.send(output.encode())

            elif received == "get_mem_info()":
                # Show the information in /proc/meminfo and send the output to the client
                output = subprocess.check_output(["cat", "/proc/meminfo"]).decode().strip()
                sock.send(output.encode())

            else:
                sock.send("Invalid command.".encode())

        sock.close()
    except ConnectionResetError:
        print("Goodbye!")
    
    except ConnectionRefusedError:
        print("\nConnection has been refused!")

if __name__=="__main__":
    client()

The command map is deliberately tight: it only exposes a handful of read-only system calls. That keeps the demo safe and predictable.

Controller: GUI Sender

The controller is a tkinter GUI with:

  • a command dropdown,
  • an IP:PORT entry,
  • a list of connected clients,
  • a send button and output area.

It listens for connections, keeps a list of sockets, and can send commands to either a selected client or all clients.

import socket
import tkinter as tk
from threading import Thread

class Controller:
    """A class to manage the GUI and handle client connections.Python GUI - tkinter - GeeksforGeeks. (2017, June 17). GeeksforGeeks. https://www.geeksforgeeks.org/python-gui-tkinter/

‌"""

    def __init__(self, master):
        """Initialize the Controller object with the GUI elements and connection data."""
        self.host = None
        self.port = None
        self.address = None
        self.socket = None
        self.list = []
        self.quit_flag = False  # flag to indicate when to quit the loop


        # Create the GUI elements
        self.master = master
        self.master.title("Controller")
        self.master.configure(bg="#EEE")

        # Create a label for the command selection
        self.command_label = tk.Label(self.master, text="Select a command:", font=("Arial", 14), bg="#EEE")
        self.command_label.grid(row=0, column=0, padx=10, pady=10, sticky="w")

        # Create a dropdown menu to select the command
        self.command_var = tk.StringVar(self.master)
        self.command_var.set("Select a command...")
        self.command_menu = tk.OptionMenu(self.master, self.command_var, "get_users()", "get_device_info()", "get_mem_info()", "get_id()")
        self.command_menu.config(width=25, font=("Arial", 12))
        self.command_menu.grid(row=0, column=1, padx=10, pady=10)

        # Create a label and an entry field for additional command parameters
        self.ip_port_label = tk.Label(self.master, text="IP:PORT", font=("Arial", 14), bg="#EEE")
        self.ip_port_label.grid(row=1, column=0, padx=10, pady=10, sticky="w")
        self.ip_port_entry = tk.Entry(self.master, width=30, font=("Arial", 12))
        self.ip_port_entry.grid(row=1, column=1, padx=10, pady=10)
        self.ip_port_entry.insert(tk.END, 'localhost:9999')

        # Create a button to send the command
        self.button = tk.Button(self.master, text="Send", command=self.controller, font=("Arial", 14))
        self.button.grid(row=0, column=2, padx=10, pady=10)

        # Create a label for the output text widget
        self.output_label = tk.Label(self.master, text="Output:", font=("Arial", 14), bg="#EEE")
        self.output_label.grid(row=2, column=0, padx=10, pady=10, sticky="w")

        # Create an output text widget with scrollbar
        self.output = tk.Text(self.master, height=10, width=50, font=("Arial", 12))
        self.output.grid(row=3, column=0, columnspan=2, padx=10, pady=10)

        self.client_box = tk.Listbox(self.master, height=9, font=("Arial", 12))
        self.client_box.grid(row=3, column=2, padx=10, pady=10)
        

        # Create a button to clear the output text widget
        self.clear_button = tk.Button(self.master, text="Clear", command=self.clear_output, font=("Arial", 14))
        self.clear_button.grid(row=4, column=0, padx=10, pady=10, sticky="w")

        # Create a button to connect to a new client
        self.connect_button = tk.Button(self.master, text="Start Server", command=self.start_bind, font=("Arial", 14))
        self.connect_button.grid(row=1, column=2, padx=10, pady=10, sticky="w")

        # Create a button to send the command to all clients
        self.send_all_button = tk.Button(self.master, text="Send All", command=self.send_all, font=("Arial", 14))
        self.send_all_button.grid(row=1, column=3, padx=10, pady=10)


    def send_all(self):
       """
        Send the selected command to all clients and display the responses in the output widget.
        """

       # Get the selected command and additional parameters from the GUI
       command = self.command_var.get()    
       # Validate the command and parameters
       if command == "Select a command...":
           self.output.insert(tk.END, "Please select a command.\n")
           return    
       responses = []
       for socket, address in self.list:
           # Send the command and parameters to each connected client
           socket.send(command.encode())    
           # Receive a response from the client and append it to the responses list
           response = socket.recv(4048).decode()
           responses.append(response)    
       # Display the responses in the output widget
       self.output.insert(tk.END, "\n".join(responses) + "\n")
            

    def start_bind(self):
        """Bind the server to the specified IP and port and start listening for clients.
        """
        self.ip_port = self.ip_port_entry.get()
        self.host = self.ip_port.split(":")[0]
        self.port = int(self.ip_port.split(":")[1])
        self.socket = socket.socket()
        self.socket.bind((self.host, self.port))
        print("Server has binded.")
        Thread(target=self.start_listening).start()

    def start_listening(self):
        """
        Listen for incoming client connections and start a new thread to handle each connection.

        """
        print("Server is listening...")
        self.socket.listen(1)
        self.output.insert(tk.END, f"Listening on {self.host} on {self.port}")
        while True:
            self.new_socket, self.address = self.socket.accept()
            self.list.append((self.new_socket, self.address))
            print("accepted client")
            self.client_box.insert(tk.END, f"{self.address[0]}:{self.address[1]}")
            thread = Thread(target=self.controller)
            thread.start()


    def controller(self):
        """
        Handle the selected command by sending it to the selected client or all clients, and display the responses
        in the output widget
        """
        # Get the selected command and additional parameters from the GUI
        command = self.command_var.get()

        # Validate the command and parameters
        if command == "Select a command...":
            self.output.insert(tk.END, "Please select a command.\n")
            return

        client_select = self.client_box.curselection()
        if client_select == ():
            # Send the command to all clients
            responses = []
            for socket, address in self.list:
                socket.send(command.encode())
                response = socket.recv(4048).decode()
                responses.append(response)
            self.output.insert(tk.END, "\n".join(responses) + "\n")
        else:
            # Send the command to the selected client
            client_select = client_select[0]
            socket = self.list[client_select][0]
            socket.send(command.encode())
            response = socket.recv(4048).decode()
            self.output.insert(tk.END, response + "\n")
        
        # Display the responses in the output widget
        
            self.output.insert(tk.END, "\n".join(responses) + "\n")




    def clear_output(self):
        """ 
        Clear the output and socket connection
        """
        self.output.delete('1.0', tk.END)
        self.socket.close()

def main():
    """
    Initialize the GUI window and start the event loop.
    """     
    # Create the GUI window
    root = tk.Tk()

    # Create an instance of the Controller class
    controller = Controller(root)

    # Start the GUI event loop
    root.mainloop()
if __name__ == "__main__":
    main()

Running It

Start the GUI first:

python3 controller.py

Then start the client in another terminal:

python3 client.py

Once the client connects, enter the IP and port (default is local) and click Start Server.

Select a command from the dropdown.

Click Send for a selected client, or Send All for every connected client.

Notes and Next Steps

This demo is intentionally limited to read-only commands. If you want to expand it, consider:

  • a command allowlist on the controller,
  • authentication for clients,
  • encrypted sockets for remote usage.

It is a clean foundation for learning about GUI control panels and socket-based command routing without overcomplicating the design.
